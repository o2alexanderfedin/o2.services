```typescript
// packages/core/src/x509.ts — the contract Plan 25-02 builds against

/** One decoded TLV node, faithful to what was read — no canonicalisation applied yet. */
export interface DerNode {
  readonly tag: number
  readonly constructed: boolean
  /** Primitive nodes only; empty for constructed nodes. */
  readonly content: Uint8Array
  /** Constructed nodes only; empty for primitive nodes. */
  readonly children: readonly DerNode[]
}

/**
 * Every refusal this module can return, across all seven obligations. Extended by
 * Plan 25-02 with the semantic kinds (unrecognised-algorithm, non-canonical-encoding
 * wired as a certificate gate, extension-too-large, too-many-extensions,
 * duplicate-extension, and the structural RFC 5280 refusals). This plan defines and
 * returns only the two kinds its own tasks produce.
 */
export type X509Failure =
  | { readonly kind: 'certificate-too-large'; readonly bytes: number; readonly limit: number }
  | { readonly kind: 'malformed-der'; readonly detail: string; readonly offset: number }
  // Plan 25-02 adds: 'unrecognised-algorithm' | 'algorithm-parameters-present'
  // | 'non-canonical-encoding' | 'extension-too-large' | 'too-many-extensions'
  // | 'duplicate-extension' | 'unrecognised-extension' | 'unique-identifier-present'
  // | 'multi-valued-rdn' | 'generalized-time-present'

export const MAX_CERTIFICATE_BYTES = 4096
export const MAX_EXTENSION_BYTES = 2048
export const MAX_EXTENSION_COUNT = 8

/** Decode one TLV tree from `bytes`, refusing anything outside the 12-tag profile. */
export function decodeDer(bytes: Uint8Array): { ok: true; node: DerNode } | { ok: false; failure: X509Failure }

/** Re-serialize a DerNode tree in this profile's one canonical form (X.690 DER). */
export function encodeDer(node: DerNode): Uint8Array

/** decode(bytes) -> encode(that tree) -> byte-compare against `bytes`. True iff identical. */
export function checkDerCanonical(bytes: Uint8Array): boolean
```

      `encodeDer` re-emits the minimal short-form length, so `checkDerCanonical`
      returns `false` on that input.
    - Given a decoded `INTEGER` node whose content carries a redundant leading `0x00`
      (high bit of the next byte already `0`), `encodeDer` strips it, so
      `checkDerCanonical` returns `false` on that input. Given an `INTEGER` whose
      leading `0x00` is NOT redundant (next byte's high bit is `1`, so the `0x00`
      disambiguates positive from negative), `encodeDer` preserves it, so
      `checkDerCanonical` returns `true`.
    - Given a decoded `BIT STRING` node that is byte-aligned (unused-bits count byte
      is `0x00`) but whose last content byte carries nonzero garbage in bit positions
      that are not part of the represented value... this case cannot occur when
      unused-bits is `0x00` (every bit is used) — instead: given a `BIT STRING` whose
      unused-bits count is nonzero (e.g. `0x03`, meaning the last 3 bits are padding)
      and those padding bits are NOT zero, `encodeDer` zeroes them on re-emit, so
      `checkDerCanonical` returns `false` on that input.
    - Given a decoded `BOOLEAN` node representing `TRUE` with content byte `0x01`
      (BER-legal, DER-illegal — DER requires exactly `0xFF`), `encodeDer` re-emits
      `0xFF`, so `checkDerCanonical` returns `false` on that input.
    - Given a fully DER-canonical input (minimal lengths throughout, no redundant
      INTEGER padding, zeroed BIT STRING padding, `0xFF` boolean), `checkDerCanonical`
      returns `true` — the "other half of the bound" case, matching
      `capability.test.ts:238-252`'s "accepts exactly at the boundary" discipline:
      without it, an `encodeDer` that always returns empty bytes would make every
      other test in this file pass for the wrong reason.
  </behavior>
  <action>
    Implement `encodeDer(node)`: recursively serialize. For constructed nodes,
    encode each child and concatenate to form the content, then emit
    `[tag, minimalLength(content), ...content]`. For primitive nodes, apply
    per-tag canonicalisation before emitting: `INTEGER` (`0x02`) — recompute the
    minimal two's-complement form from the semantic value (strip a redundant leading
    `0x00` when the following byte's high bit is 0, or a redundant leading `0xFF` when
    the following byte's high bit is 1, for a value that was already read as
    non-negative in this profile — `serialNumber` is this profile's only `INTEGER`
    field that can appear standalone; the `version` INTEGER is always a small
    non-negative literal). `BIT STRING` (`0x03`) — read the first content byte as the
    unused-bits count, mask that many low bits of the last content byte to `0`, emit
    unchanged unused-bits count and the rest of the content verbatim. `BOOLEAN`
    (`0x01`) — emit `0x00` for a zero content byte, `0xFF` for any nonzero content
    byte. All other tags — emit content bytes verbatim (their canonicalisation is
    handled structurally, not per-byte: multi-valued `SET`s are refused outright by
    Plan 25-02 rather than sorted, per RESEARCH.md §2 row 5, so `encodeDer` never
    needs `SET OF` ordering logic).

    belongs with the obligation it satisfies (X509-03) in the same commit as that
    obligation's test.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| bytes → `decodeDer` | Attacker-supplied bytes (a certificate presented by any peer) cross into a recursive-descent parser here |
| `DerNode` tree → `encodeDer` | Internal boundary; content this module itself decoded, re-serialized for comparison |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-25-01 | Denial of Service | `decodeDer` (indefinite-length / recursive-length parsing) | mitigate | Indefinite length (`0x80`) refused at the length-octet check, before any recursive descent begins — Task 2 |
| T-25-02 | Denial of Service | `preParseCheck` (certificate/extension bomb via oversized input) | mitigate | `MAX_CERTIFICATE_BYTES` checked before `decodeDer` is called at all — Task 1 constant, Task 2 wiring |
| T-25-03 | Tampering | `decodeDer` (BER/DER confusion — accepting a non-canonical encoding as canonical) | mitigate | `checkDerCanonical`'s re-encode/byte-compare round trip, covering all five known divergence classes with one mechanism — Task 3 |
| T-25-04 | Tampering | `decodeDer` (unbounded recursion depth from a deeply nested constructed TLV) | accept | Bounded transitively by `MAX_CERTIFICATE_BYTES`: each recursion level consumes at least 2 bytes (tag + length), so recursion depth is capped at half the certificate size ceiling (2048 levels worst case) — not separately bounded in this plan; Plan 25-02's structural refusals (fixed field positions, no open-ended nesting) further reduce the reachable depth to a small constant in practice |
</threat_model>
